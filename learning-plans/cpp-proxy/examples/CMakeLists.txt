cmake_minimum_required(VERSION 3.28)
project(proxy_learning LANGUAGES CXX)
set(PROXY_SOURCE_DIR "" CACHE PATH "Path to the official Proxy 4.1.0 checkout")
if(NOT EXISTS "${PROXY_SOURCE_DIR}/include/proxy/proxy.h")
  message(FATAL_ERROR "Set -DPROXY_SOURCE_DIR=/path/to/proxy-4.1.0; see README.md")
endif()
add_subdirectory("${PROXY_SOURCE_DIR}" "${CMAKE_CURRENT_BINARY_DIR}/proxy")
enable_testing()
foreach(example IN ITEMS 01-basic 02-ownership 03-expressions 04-erasure-model)
  add_executable(${example} ${example}.cpp)
  target_compile_features(${example} PRIVATE cxx_std_20)
  target_link_libraries(${example} PRIVATE msft_proxy4::proxy)
  add_test(NAME ${example} COMMAND ${example})
endforeach()
